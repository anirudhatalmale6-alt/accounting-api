      vat_service.dart          -- VAT return + HMRC submission service
  presentation/

MENU BUTTONS (add to your drawer/menu):
----------------------------------------

import 'core/auth/role_permissions.dart';

// Jobs menu (visible to owner, admin, engineer)
if (RolePermissions.canAccessJobs(userRole))
  ListTile(
    leading: const Icon(Icons.calendar_month),
    title: const Text("Jobs"),
    onTap: () {
      Navigator.pushNamed(context, "/jobs/calendar");
    },
  );

// Today's jobs
if (RolePermissions.canAccessJobs(userRole))
  ListTile(
    leading: const Icon(Icons.today),
    title: const Text("Today's Jobs"),
    onTap: () {
      Navigator.pushNamed(context, "/jobs/today");
    },
  );

// Engineers (visible to owner, admin)
if (RolePermissions.canManageTeam(userRole))
  ListTile(
    leading: const Icon(Icons.engineering),
    title: const Text("Engineers"),
    onTap: () {
      Navigator.pushNamed(context, "/engineers");
    },
  );

// Live board
ListTile(
  leading: const Icon(Icons.dashboard_customize),
  title: const Text("Engineer Live Board"),
  onTap: () => Navigator.pushNamed(context, "/dispatch/live-board"),

// Reports (visible to owner, admin, accountant)

      Navigator.pushNamed(context, "/reports");

CHAT INTEGRATION:
-----------------

Add imports:
  import 'presentation/screens/chat/chat_list_screen.dart';
  import 'presentation/screens/chat/chat_conversation_screen.dart';

Add route:
  if (settings.name == "/chat") {
    return MaterialPageRoute(
      builder: (_) => const ChatListScreen(),
    );
  }

Add menu item (all roles can access chat):
  ListTile(
    leading: const Icon(Icons.chat),
    title: const Text("Chat"),
    onTap: () {
      Navigator.pushNamed(context, "/chat");
    },
  );

IMPORTANT - save userId on login:
  After successful login, save the user ID to SharedPreferences so the chat
  screen can identify which messages are yours:

    final prefs = await SharedPreferences.getInstance();
    await prefs.setInt("userId", loginResponse["user"]["id"]);

API Endpoints:
  GET  /chat/group           - Group messages (paginated, ?before=id&limit=50)
  POST /chat/group           - Send group message (body: {"message": "..."})
  GET  /chat/direct/:userId  - Direct messages with user (paginated)
  POST /chat/direct/:userId  - Send direct message (body: {"message": "..."})
  GET  /chat/contacts        - List team members for DM with last message preview


CSV IMPORT (Customers & Suppliers):
------------------------------------

Add import:
  import 'presentation/screens/import/csv_import_screen.dart';

Add routes:
  if (settings.name == "/customers/import") {
    return MaterialPageRoute(
      builder: (_) => const CsvImportScreen(type: "customers"),
    );
  }
  if (settings.name == "/suppliers/import") {
    return MaterialPageRoute(
      builder: (_) => const CsvImportScreen(type: "suppliers"),
    );
  }

Add import buttons on Customers/Suppliers list screens:
  IconButton(
    icon: const Icon(Icons.upload_file),
    onPressed: () => Navigator.pushNamed(context, "/customers/import"),
  )

Dependency: add file_picker to pubspec.yaml:
  file_picker: ^8.0.0

API Endpoints:
  POST /customers/import  - body: {"rows": [{"name":"...", "email":"...", ...}]}
  POST /suppliers/import  - body: {"rows": [{"name":"...", "email":"...", ...}]}

CSV columns: name (required), email, phone, address, vat_number, contact_person
